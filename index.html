async function loadRooms() { const roomsContainer = document.getElementById("roomsContainer"); try { const response = await fetch("/rooms"); if (!response.ok) { throw new Error("Failed to load rooms"); } const rooms = await response.json(); // Display Room 207 only const room207 = rooms.filter(room => String(room.id ?? room.room ?? room.roomNumber).trim() === "207" ); roomsContainer.innerHTML = ""; if (room207.length === 0) { roomsContainer.innerHTML = '

Room 207 was not found.

'; return; } room207.forEach(room => { const card = document.createElement("div"); card.className = "room-card"; card.innerHTML = `

Room 207

Patient Monitoring

`; card.addEventListener("click", () => { openRoom(room.id ?? room.room ?? room.roomNumber); }); roomsContainer.appendChild(card); }); } catch (error) { console.error("Error loading rooms:", error); roomsContainer.innerHTML = '

Unable to load Room 207.

'; } }
